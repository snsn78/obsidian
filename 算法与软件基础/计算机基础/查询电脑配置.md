```bat
Write-Host "`n========== 1. Windows 版本 ==========" -ForegroundColor Cyan
(Get-ComputerInfo).WindowsProductName; (Get-ComputerInfo).OsVersion

Write-Host "`n========== 2. GPU 信息 ==========" -ForegroundColor Cyan
try { nvidia-smi --query-gpu=name,driver_version,memory.total --format=csv 2>$null } catch { Write-Host "⚠ nvidia-smi 不可用" }

Write-Host "`n========== 3. CUDA (nvcc) ==========" -ForegroundColor Cyan
try { nvcc --version 2>$null } catch { Write-Host "⚠ nvcc 不在 PATH 中" }
Write-Host "---"
if (Test-Path "C:\Program Files\NVIDIA GPU Computing Toolkit\CUDA") { Get-ChildItem "C:\Program Files\NVIDIA GPU Computing Toolkit\CUDA" } else { Write-Host "⚠ 未找到 CUDA 安装目录" }

Write-Host "`n========== 4. Miniconda3 ==========" -ForegroundColor Cyan
try { conda --version 2>$null } catch { Write-Host "⚠ conda 不在 PATH 中" }
Write-Host "---"
$condaDirs = @("$env:USERPROFILE\miniconda3", "$env:USERPROFILE\anaconda3", "C:\ProgramData\miniconda3", "C:\ProgramData\anaconda3")
foreach ($d in $condaDirs) { if (Test-Path $d) { Write-Host "发现目录: $d" } }

Write-Host "`n========== 5. Python ==========" -ForegroundColor Cyan
try { python --version 2>$null } catch { Write-Host "⚠ python 不在 PATH 中" }

Write-Host "`n========== 6. PyTorch GPU 可用性 ==========" -ForegroundColor Cyan
python -c "import torch; print('PyTorch:', torch.__version__); print('CUDA 可用:', torch.cuda.is_available()); print('GPU 数量:', torch.cuda.device_count()); print('GPU:', torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'N/A')" 2>$null
if ($LASTEXITCODE -ne 0) { Write-Host "⚠ PyTorch 未安装或无法导入" }

Write-Host "`n========== 7. WSL 发行版 ==========" -ForegroundColor Cyan
wsl --list --verbose 2>$null

Write-Host "`n========== 检查完毕 ==========" -ForegroundColor Cyan
```
