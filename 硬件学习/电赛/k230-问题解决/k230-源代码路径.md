"""
CanMV K230 YOLOv8 steel-ball tracking and calibrated position measurement.

Yahboom K230 screen adapted version.
亚博 K230 屏幕适配版：
1. 使用 Sensor()，不强制 Sensor(id=2)。
2. Display.init(Display.ST7701, to_ide=True)，不传 width/height，避免 IDE 绿一下后黑屏。
3. 不使用 Display.bind_layer，也不使用 OSD layer；直接在摄像头图像上画框后 Display.show_image(img)。
4. CAM_CHN_ID_0: RGB565 640x480，用于屏幕显示。
5. CAM_CHN_ID_2: RGB888_PLANAR 320x240，用于 YOLO 输入。

Model location on the board:
    /sdcard/kmodel/model.kmodel
人话：sdcard 目录下创建 kmodel 文件夹，把 model.kmodel 放进去。

注意：下面 CAL_PX 已从原 800 宽等比例缩放到 640 宽，能先跑通；
      真要精确测量，摄像头固定后请重新标定 CAL_PX。
"""
import os
import gc
import time
import image
import nncase_runtime as nn

from media.sensor import *
from media.display import *
from media.media import *
from libs.YOLO import YOLOv8


# ==================== 用户可调参数区 ====================
KMODEL_PATH = "/sdcard/kmodel/model.kmodel"
LABELS = ["ball"]

# 亚博 K230 常用 LCD / CanMV IDE 预览尺寸
DISPLAY_WIDTH = 640
DISPLAY_HEIGHT = 480

# True：板载 LCD + CanMV IDE 右侧预览一起显示
# False：通常只走板载 LCD，若 IDE 预览异常可改 False 试试
DISPLAY_TO_IDE = True

# YOLO 输入通道尺寸，宽度 320 已经是 16 对齐
RGB888P_SIZE = [320, 240]
MODEL_INPUT = [640, 640]

CONF_THRESH = 0.05
NMS_THRESH = 0.45
MAX_BOXES = 15

# 只搜索横杆附近区域。摄像头固定后可按实际画面微调这四个比例。
ROI_X1 = int(DISPLAY_WIDTH * 0.00)
ROI_Y1 = int(DISPLAY_HEIGHT * 0.35)
ROI_X2 = int(DISPLAY_WIDTH * 1.00)
ROI_Y2 = int(DISPLAY_HEIGHT * 0.65)

# 过滤离谱检测框。摄像头更低/球更大时可增大 MAX_BOX_PX。
MIN_BOX_PX = 8
MAX_BOX_PX = 130
MIN_ASPECT = 0.45
MAX_ASPECT = 2.20

# 单目标跟踪和平滑参数
TRACK_MAX_JUMP_PX = 145
TRACK_RESET_FRAMES = 8
FILTER_ALPHA = 0.65

# 固定补光后再设置 3000~10000；0 表示保持自动曝光
MANUAL_EXPOSURE_US = 0

# ==================== 一维标定表 ====================
# 原代码的 CAL_PX 是基于 800x480 屏幕坐标。
# 这里已按 640/800 等比例缩放，先保证亚博 640x480 屏幕模式能跑。
# 精确比赛测量时，请重新采集 RAW cx 后替换本表。
CAL_CM = (
    -12.0, -11.5, -11.0, -10.5, -10.0, -9.5, -9.0, -8.5, -8.0,
    -7.5, -7.0, -6.5, -6.0, -5.5, -5.0, -4.5, -4.0, -3.5, -3.0,
    -2.5, -2.0, -1.5, -1.0, -0.5, 0.0,
    0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 4.5, 5.0, 5.5,
    6.0, 6.5, 7.0, 7.5, 8.0, 8.5, 9.0, 9.5, 10.0, 10.5,
    11.0, 11.5, 12.0
)
CAL_PX = (
    33, 44, 56, 68, 80, 92, 104, 116, 128,
    141, 153, 165, 178, 190, 202, 214, 226, 239, 251,
    263, 274, 286, 298, 310, 321,
    329, 338, 346, 357, 366, 378, 389, 401, 412, 425, 437,
    450, 462, 476, 489, 502, 515, 528, 541, 554, 566,
    578, 590, 602
)
CALIBRATED = True

sensor = None
yolo = None


def axis_map(value, src, dst):
    """分段线性映射：像素 <-> 厘米，用来补偿一维畸变。"""
    if value <= src[0]:
        i = 0
    elif value >= src[-1]:
        i = len(src) - 2
    else:
        i = 0
        while value > src[i + 1]:
            i += 1

    span = src[i + 1] - src[i]
    if span == 0:
        return dst[i]
    return dst[i] + (value - src[i]) * (dst[i + 1] - dst[i]) / span


def pixel_to_cm(px):
    return axis_map(px, CAL_PX, CAL_CM)


def cm_to_pixel(cm):
    return int(axis_map(cm, CAL_CM, CAL_PX) + 0.5)


def draw_scale(img, ball_px=None):
    """在画面底部画 -5cm 到 +5cm 标尺。"""
    base_y = DISPLAY_HEIGHT - 3
    left_x = cm_to_pixel(-5.0)
    right_x = cm_to_pixel(5.0)

    img.draw_line(left_x, base_y, right_x, base_y,
                  color=(255, 255, 255), thickness=2)

    for half_cm in range(-10, 11):
        cm = half_cm * 0.5
        px = cm_to_pixel(cm)
        is_major = (half_cm % 2 == 0)
        tick_h = 26 if half_cm == 0 else (18 if is_major else 10)
        tick_color = (0, 255, 255) if half_cm == 0 else (255, 255, 255)

        img.draw_line(px, base_y, px, base_y - tick_h,
                      color=tick_color,
                      thickness=2 if is_major else 1)

        if is_major:
            label = "O" if half_cm == 0 else ("%+d" % int(cm))
            text_x = max(0, min(DISPLAY_WIDTH - 34, px - 14))
            img.draw_string_advanced(text_x, base_y - 48, 18, label,
                                     color=tick_color)

    if ball_px is not None:
        marker_x = max(0, min(DISPLAY_WIDTH - 1, int(ball_px)))
        img.draw_circle(marker_x, base_y - 23, 5,
                        color=(255, 255, 0), thickness=3)


def camera_init():
    """亚博 K230 稳定显示初始化。"""
    global sensor

    nn.shrink_memory_pool()

    # 亚博板这里用 Sensor()，不要强制 Sensor(id=2)
    sensor = Sensor()
    sensor.reset()
    sensor.set_vflip(True)
    sensor.set_hmirror(True)

    # 通道0：给屏幕显示，RGB565 可以直接画图和 show_image
    sensor.set_framesize(width=DISPLAY_WIDTH, height=DISPLAY_HEIGHT,
                         chn=CAM_CHN_ID_0)
    sensor.set_pixformat(Sensor.RGB565, chn=CAM_CHN_ID_0)

    # 通道2：给 YOLO 输入，必须保持 RGB888_PLANAR
    sensor.set_framesize(width=RGB888P_SIZE[0], height=RGB888P_SIZE[1],
                         chn=CAM_CHN_ID_2)
    sensor.set_pixformat(PIXEL_FORMAT_RGB_888_PLANAR,
                         chn=CAM_CHN_ID_2)

    if MANUAL_EXPOSURE_US > 0:
        try:
            sensor.auto_exposure(False)
        except:
            try:
                sensor.set_auto_exposure(False)
            except:
                pass

    # 关键改动：亚博屏幕/IDE 模式不要塞 width/height，也不要 bind_layer
    Display.init(Display.ST7701, to_ide=DISPLAY_TO_IDE)

    MediaManager.init()
    sensor.run()

    if MANUAL_EXPOSURE_US > 0:
        try:
            sensor.exposure(MANUAL_EXPOSURE_US)
        except:
            pass


def camera_deinit():
    """按亚博板更稳的顺序释放资源。"""
    global sensor

    if sensor is not None:
        try:
            sensor.stop()
        except:
            pass
    try:
        Display.deinit()
    except:
        pass
    try:
        os.exitpoint(os.EXITPOINT_ENABLE_SLEEP)
    except:
        pass
    time.sleep_ms(100)
    try:
        MediaManager.deinit()
    except:
        pass


def is_inside_roi(cx, cy):
    return ROI_X1 <= cx <= ROI_X2 and ROI_Y1 <= cy <= ROI_Y2


def select_ball(boxes, scores, previous_x):
    """选择一个最像钢球的目标，优先保持跟踪连续性。"""
    best = None
    best_rank = -1000000.0
    valid_count = 0

    for i in range(len(boxes)):
        x, y, w, h = map(int, boxes[i])
        score = float(scores[i])

        if w <= 0 or h <= 0:
            continue
        if w < MIN_BOX_PX or h < MIN_BOX_PX:
            continue
        if w > MAX_BOX_PX or h > MAX_BOX_PX:
            continue

        aspect = float(w) / float(h)
        if aspect < MIN_ASPECT or aspect > MAX_ASPECT:
            continue

        cx = x + (w >> 1)
        cy = y + (h >> 1)
        if not is_inside_roi(cx, cy):
            continue

        valid_count += 1
        if previous_x is None:
            rank = score * 100.0
        else:
            jump = abs(cx - previous_x)
            if jump > TRACK_MAX_JUMP_PX and score < 0.50:
                continue
            rank = score * 100.0 - jump * 0.20

        if rank > best_rank:
            best_rank = rank
            best = (x, y, w, h, cx, cy, score)

    return best, valid_count


def main():
    global sensor, yolo

    os.exitpoint(os.EXITPOINT_ENABLE)
    try:
        file_size = os.stat(KMODEL_PATH)[6]
        print("kmodel: %d bytes OK" % file_size)
    except OSError:
        print("[ERROR] model not found: " + KMODEL_PATH)
        return

    camera_is_init = False
    try:
        camera_init()
        camera_is_init = True

        yolo = YOLOv8(
            task_type="detect",
            mode="video",
            kmodel_path=KMODEL_PATH,
            labels=LABELS,
            rgb888p_size=RGB888P_SIZE,
            model_input_size=MODEL_INPUT,
            display_size=[DISPLAY_WIDTH, DISPLAY_HEIGHT],
            conf_thresh=CONF_THRESH,
            nms_thresh=NMS_THRESH,
            max_boxes_num=MAX_BOXES,
            debug_mode=0
        )
        yolo.config_preprocess()

        fps = 0.0
        fps_frames = 0
        fps_time = time.ticks_ms()
        print_divider = 0

        last_raw_x = None
        filtered_x = None
        raw_history = []
        lost_frames = TRACK_RESET_FRAMES
        roi_w = ROI_X2 - ROI_X1
        roi_h = ROI_Y2 - ROI_Y1

        while True:
            os.exitpoint()

            # 通道2只用于 YOLO，不拿它直接显示
            ai_frame = sensor.snapshot(chn=CAM_CHN_ID_2)
            input_np = ai_frame.to_numpy_ref()
            result = yolo.run(input_np)

            # 通道0用于屏幕显示：直接在摄像头图像上画框/文字/标尺
            disp_img = sensor.snapshot(chn=CAM_CHN_ID_0)
            disp_img.draw_rectangle(ROI_X1, ROI_Y1, roi_w, roi_h,
                                    color=(255, 255, 0), thickness=2)

            selected = None
            valid_count = 0
            if result and len(result) > 2 and result[0] is not None:
                selected, valid_count = select_ball(
                    result[0], result[2], last_raw_x)

            if selected is not None:
                x, y, w, h, raw_cx, cy, score = selected
                lost_frames = 0

                raw_history.append(raw_cx)
                if len(raw_history) > 3:
                    raw_history.pop(0)
                median_cx = sorted(raw_history)[len(raw_history) // 2]

                if filtered_x is None:
                    filtered_x = float(median_cx)
                else:
                    filtered_x += FILTER_ALPHA * (median_cx - filtered_x)
                last_raw_x = int(filtered_x)

                position_cm = pixel_to_cm(filtered_x)
                position_prefix = "X:" if CALIBRATED else "X~"

                disp_img.draw_rectangle(x, y, w, h,
                                        color=(0, 255, 0), thickness=2)
                disp_img.draw_cross(int(filtered_x), cy,
                                    color=(255, 0, 0),
                                    size=12, thickness=2)
                disp_img.draw_string_advanced(
                    x, max(5, y - 22), 20,
                    "%s%+.2fcm %.2f" %
                    (position_prefix, position_cm, score),
                    color=(0, 255, 0)
                )
                draw_scale(disp_img, filtered_x)

                print_divider += 1
                if print_divider >= 10:
                    print("RAW cx=%d | MEDIAN cx=%d | FILTER cx=%d | "
                          "X=%+.3f cm | conf=%.3f" %
                          (raw_cx, median_cx, int(filtered_x),
                           position_cm, score))
                    print_divider = 0
            else:
                lost_frames += 1
                if lost_frames >= TRACK_RESET_FRAMES:
                    last_raw_x = None
                    filtered_x = None
                    raw_history = []
                draw_scale(disp_img, None)

            fps_frames += 1
            if fps_frames >= 10:
                now = time.ticks_ms()
                elapsed = time.ticks_diff(now, fps_time) / 1000.0
                if elapsed > 0:
                    fps = fps_frames / elapsed
                fps_time = now
                fps_frames = 0
                gc.collect()

            disp_img.draw_string_advanced(
                15, 10, 22, "FPS: %.1f" % fps, color=(255, 255, 0))
            disp_img.draw_string_advanced(
                15, 36, 22, "Candidates: %d" % valid_count,
                color=(0, 255, 0) if selected is not None else (255, 0, 0))
            disp_img.draw_string_advanced(
                DISPLAY_WIDTH - 125, 10, 18,
                "CAL: OK" if CALIBRATED else "CAL: TEMP",
                color=(0, 255, 0) if CALIBRATED else (255, 128, 0))

            # 关键改动：不用 OSD layer，直接显示整张图
            Display.show_image(disp_img)
            gc.collect()

    except KeyboardInterrupt:
        print("User interrupted.")
    except BaseException as e:
        print("Exception:", e)
    finally:
        if yolo is not None:
            try:
                yolo.deinit()
            except:
                pass
        if camera_is_init:
            camera_deinit()
        gc.collect()


if __name__ == "__main__":
    main()
